# Prussia NGF Germany
REPLACE_OR_CREATE:je_the_matter_of_the_reichsland = {
    activation_description = { always = yes }

    group = je_group_historical_content
    icon = "gfx/interface/icons/event_icons/waving_flag.dds"

    is_shown_when_inactive = {
        OR = {
            c:NGF ?= root
            c:GER ?= root
        }
    }

    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
            owns_entire_state_region = STATE_ALSACE_LORRAINE
        }
        effect = {
            add_loyalists = {
                value = 0.05
            }
            add_modifier = {
                name = kmpmp_the_reichsland_buff
                years = 5
            }
            add_modifier = {
                name = kmpmp_the_reichsland_training_buff
                years = 1
            }
        }
    }

    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            year >= 1871
        }
    }

}

#FRANCE:
REPLACE_OR_CREATE:je_the_border_with_the_germans = {
    activation_description = { always = yes }

    group = je_group_historical_content
    icon = "gfx/interface/icons/event_icons/waving_flag.dds"

    is_shown_when_inactive = {
        c:FRA ?= root
    }

    possible = {
        year >= 1845
    }

    immediate = {
        if = {
            limit = {
                state_region_owned_and_incorporated = { STATE_REGION = STATE_ALSACE_LORRAINE }
            }
            add_modifier = {
                name = kmpmp_historical_border_buff
            }
        }
    }

    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
            year >= 1871
        }
        effect = {
            if = {
                limit = {
                    has_modifier = kmpmp_historical_border_buff
                }
                remove_modifier = kmpmp_historical_border_buff
            }
        }
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            NOT = {
                owns_entire_state_region = STATE_ALSACE_LORRAINE
            }
        }
        effect = {
            if = {
                limit = {
                    has_modifier = kmpmp_historical_border_buff
                }
                remove_modifier = kmpmp_historical_border_buff
            }
        }
    }

}

REPLACE_OR_CREATE:je_humiliated_by_the_germans = {
    activation_description = { always = yes }

    group = je_group_historical_content
    icon = "gfx/interface/icons/event_icons/waving_flag.dds"

    is_shown_when_inactive = {
        # added through on_actions if STATE_ALSACE_LORRAINE is lost
        # c:FRA ?= root
    }

    immediate = {
        set_variable = {
            name = je_humiliated_by_the_germans_activated_years
            value = 0
        }
    }

    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
            owns_entire_state_region = STATE_ALSACE_LORRAINE
        }
        effect = {

        }
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            scope:journal_entry = {
                is_goal_complete = yes
            }
            NOT = {
                owns_entire_state_region = STATE_ALSACE_LORRAINE
            }
        }
        effect = {
            add_modifier = {
                name = kmpmp_humiliated_by_the_germans
            }
        }
    }

    # on_action which is triggered every first day of the year
	on_yearly_pulse = {
		effect = {
            change_variable = {
                name = je_humiliated_by_the_germans_activated_years
                add = 1
            }
		}
	}

    # a script value computing the goal completion metric
	current_value = {
		value = var:je_humiliated_by_the_germans_activated_years
	}

	# when the journal entry is activated current_value and goal_add_value are evaluated and added together to determine the goal value
	goal_add_value = {
		value = 7
	}

    progressbar = yes
}
