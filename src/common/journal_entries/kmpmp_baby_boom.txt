REPLACE_OR_CREATE:kmpmp_je_baby_boom = {
    activation_description = { always = yes }

    # Sweden, Portugal, Netherlands, Bolivia, Brazil, Mexico add babyboom JE
    # req: has invented pharmaceuticals, has SOL of 10 avg on nation
    # grants: 10% birthrate for 10 years

    icon = "gfx/interface/icons/invention_icons/pharmaceuticals.dds"
    group = je_group_technology

    is_shown_when_inactive = {
        NOT = {
            has_variable = kmpmp_je_baby_boom_completed
        }
        OR = {
            c:SWE ?= ROOT # sweden
            c:SCA ?= ROOT # scandinavia
            c:POR ?= ROOT # portugal
            c:NET ?= ROOT # netherlands
            c:UNL ?= ROOT # united netherlands
            c:BOL ?= ROOT # bolivia
            c:FND ?= ROOT # federation des andes
			c:PEU ?= ROOT # peru
			c:PBC ?= ROOT # peru-bolivia
            c:BRZ ?= ROOT # brazil
            c:MEX ?= ROOT # mexico
        }
    }

    possible = {
        has_technology_researched = pharmaceuticals
    }

    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
            hidden_trigger = { # workaround to fix je not triggering on_complete if it's already completable when activated
                scope:journal_entry = {
                    journal_entry_age > 0
                }
            }

            has_technology_researched = pharmaceuticals
            average_sol >= 10
        }
        effect = {
            set_variable = kmpmp_je_baby_boom_completed

            add_modifier = {
                name = kmpmp_baby_boom_modifier
                years = 5 #10
            }
        }
    }

}
