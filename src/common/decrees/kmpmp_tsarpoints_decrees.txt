REPLACE:decree_promote_national_values_tsarpoints = {
	texture = "gfx/interface/icons/decree/decree_promote_national_values.dds"
	unlocking_technologies = {
		# KMPMP unlocks the base decree with Centralization.
		centralization
	}

	country_trigger = {
		custom_tooltip = {
			text = tsarpoints_decree_upgraded_tt
			OR = {
				has_variable = tsarpoints_upgraded_promote_national_values
				overlord ?= { has_variable = tsarpoints_upgraded_promote_national_values }
			}
		}
		NOT = { has_law_or_variant = law_type:law_multicultural }
	}

	state_trigger = {
		NOT = { has_decree = decree_promote_national_values }
	}

	modifier = {
		# Keep the KMPMP base of 2.0 and the vanilla upgrade of 0.15.
		state_assimilation_mult = 2.15
		state_conversion_mult = 2.15
		state_allow_assimilation_in_homeland_bool = yes
		state_allow_assimilation_without_presence_bool = yes
		state_allow_conversion_without_presence_bool = yes
		state_radicalism_increases_violent_hostility_mult = 0.65
		state_radicalism_increases_cultural_erasure_mult = 0.5
		state_radicalism_increases_open_prejudice_mult = 0.375
		# Retain the base decree's loyalism bonuses when it is upgraded.
		state_loyalism_increases_full_acceptance_mult = 0.25
		state_loyalism_increases_second_rate_citizen_mult = 0.25
	}
	cost = 125

	ai_weight = {
		value = 0

		if = {
			limit = {
				scope:country = {
					NOR = {
						has_law_or_variant = law_type:law_multicultural
						has_law_or_variant = law_type:law_subjecthood
					}
				}
				primary_cultures_percent_state = {
					target_country = owner
					value <= 0.9
				}
			}

			if = { # Colonies
				limit = {
					scope:country = {
						is_not_in_geographic_region = geographic_region_europe
						any_primary_culture = {
							has_discrimination_trait_group = heritage_group_european
						}
					}
				}
				add = 300
			}
			else_if = {
				limit = {
					scope:country = {
						OR = {
							has_strategy = ai_strategy_reactionary_agenda
							has_strategy = ai_strategy_conservative_agenda
						}
					}
				}
				add = 100
			}
			else_if = {
				limit = {
					scope:country = {
						has_strategy = ai_strategy_nationalist_agenda
					}
				}
				add = 200
			}


			if = {
				limit = {
					primary_cultures_percent_state = {
						target_country = owner
						value <= 0.1
					}
				}
				multiply = 5.0
			}
			else_if = {
				limit = {
					primary_cultures_percent_state = {
						target_country = owner
						value <= 0.3
					}
				}
				multiply = 3.0
			}
			else_if = {
				limit = {
					primary_cultures_percent_state = {
						target_country = owner
						value <= 0.5
					}
				}
				multiply = 2.0
			}

			if = {
				limit = {
					is_homeland_of_country_cultures = scope:country
				}
				multiply = 5.0
			}
		}

		if = {
			limit = {
				OR = {
					NOT = { scope:country = { has_law_or_variant = law_type:law_total_separation } }
					AND = {
						scope:country = { is_not_in_geographic_region = geographic_region_africa }
						is_in_geographic_region = geographic_region_africa
					}
				}
				religion_percent_state = {
					target = scope:country.religion
					value <= 0.9
				}
			}

			if = {
				limit = {
					scope:country = {
						has_strategy = ai_strategy_conservative_agenda
					}
				}
				add = 50
			}
			else_if = {
				limit = {
					scope:country = {
						has_strategy = ai_strategy_reactionary_agenda
					}
				}
				add = 100
			}

			if = {
				limit = {
					scope:country = {
						OR = {
							has_law_or_variant = law_type:law_state_religion
							has_law_or_variant = law_type:law_state_atheism
						}
					}
				}
				add = 50
			}

			if = {
				limit = {
					religion_percent_state = {
						target = scope:country.religion
						value <= 0.1
					}
				}
				multiply = 5.0
			}
			else_if = {
				limit = {
					religion_percent_state = {
						target = scope:country.religion
						value <= 0.3
					}
				}
				multiply = 3.0
			}
			else_if = {
				limit = {
					religion_percent_state = {
						target = scope:country.religion
						value <= 0.5
					}
				}
				multiply = 2.0
			}
		}

		if = {
			limit = {
				state_population < 100000
			}
			multiply = 0
		}
		else_if = {
			limit = {
				state_population > 300000
			}
			multiply = 2.0
		}
		else_if = {
			limit = {
				scope:country = {
					OR = {
						has_law = law_type:law_cultural_exclusion
						has_law = law_type:law_freedom_of_conscience
					}
				}
			}
			multiply = 0.5
		}
		else_if = {
			limit = {
				scope:country = {
					OR = {
						has_law = law_type:law_multicultural
						has_law = law_type:law_total_separation
					}
				}
			}
			multiply = 0.1
		}

		if = {
			limit = {
				scope:country = {
					has_journal_entry = je_russification
				}
				culture_percent_state = {
					target = cu:russian
					value < je_russification_decree_russian_culture_target # set in event_values
				}
				OR = {
					is_in_geographic_region = geographic_region_finland_old
					is_in_geographic_region = geographic_region_baltic_states_old
					is_in_geographic_region = geographic_region_poland_old
				}
			}
			add = 250
		}
		if = {
			limit = {
				scope:country = {
					has_journal_entry = je_tp_pool_cultural_dominance
				}
				culture_percent_state = {
					target = cu:russian
					value < 0.5
				}
			}
			add = 250
		}
		if = {
			limit = {
				scope:country = {
					has_variable = ai_ukraine_renaissance_repress
				}
				state_region = { is_homeland = cu:ukrainian }
			}
			add = 250
		}
		if = {
			limit = {
				scope:country = {
					has_variable = ai_ukraine_renaissance_moderate
					cultural_renaissance_fervor_is_high = { CULTURE = ukrainian }
				}
				state_region = { is_homeland = cu:ukrainian }
			}
			add = 150
		}
		if = {
			limit = {
				scope:country = {
					has_variable = ai_belarus_renaissance_repress
				}
				state_region = { is_homeland = cu:byelorussian }
			}
			add = 250
		}
		if = {
			limit = {
				scope:country = {
					has_variable = ai_belarus_renaissance_moderate
					cultural_renaissance_fervor_is_high = { CULTURE = byelorussian }
				}
				state_region = { is_homeland = cu:byelorussian }
			}
			add = 150
		}
		if = {
			limit = {
				scope:country = {
					has_journal_entry = je_tp_tsar_gendarme_of_europe
					NOT = { has_variable = je_tp_tsar_gendarme_of_europe_button_3_complete }
				}
				OR = {
					AND = {
						is_in_geographic_region = geographic_region_diocese_of_kamchatka
						NOT = {
							religion_percent_state = {
								target = rel:orthodox
								value >= 0.75
							}
						}
					}
					AND = {
						is_in_geographic_region = geographic_region_baltic_governorates
						NOT = {
							religion_percent_state = {
								target = rel:orthodox
								value >= 0.25
							}
						}
					}
				}
			}
			add = 250
		}

		if = {
			limit = {
				primary_cultures_percent_state = {
					target_country = owner
					value > 0.9
				}
				religion_percent_state = {
					target = scope:country.religion
					value > 0.9
				}
			}
			multiply = 0
		}
		else_if = {
			limit = {
				primary_cultures_percent_state = {
					target_country = owner
					value > 0.5
				}
				religion_percent_state = {
					target = scope:country.religion
					value > 0.5
				}
			}
			multiply = 0.5
		}

		if = {
			limit = {
				scope:country = {
					has_variable = ai_ukraine_renaissance_moderate
				}
				state_region = { is_homeland = cu:ukrainian }
				NOT = { cultural_renaissance_fervor_is_high = { CULTURE = ukrainian } }
			}
			multiply = 0.5
		}
		if = {
			limit = {
				scope:country = {
					has_variable = ai_ukraine_renaissance_accelerate
				}
				state_region = { is_homeland = cu:ukrainian }
			}
			multiply = 0.1
		}
		if = {
			limit = {
				scope:country = {
					has_variable = ai_belarus_renaissance_moderate
				}
				state_region = { is_homeland = cu:byelorussian }
				NOT = { cultural_renaissance_fervor_is_high = { CULTURE = byelorussian } }
			}
			multiply = 0.5
		}
		if = {
			limit = {
				scope:country = {
					has_variable = ai_belarus_renaissance_accelerate
				}
				state_region = { is_homeland = cu:byelorussian }
			}
			multiply = 0.1
		}
	}
}

REPLACE:decree_greener_grass_campaign_tsarpoints = {

			text = tsarpoints_decree_upgraded_tt
			OR = {
				has_variable = tsarpoints_upgraded_greener_grass_campaign
				overlord ?= { has_variable = tsarpoints_upgraded_greener_grass_campaign }
			}
		}
	}

	state_trigger = {
		NOT = { has_decree = decree_greener_grass_campaign }
		is_isolated_from_market = no
		# Both Greener Grass variants exclude the KMPMP Emmigration Edict.
		NOT = { has_decree = kmpmp_decree_emmigration_edict }
	}
	modifier = {
		state_migration_pull_add = 25
		state_migration_pull_mult = 0.3
	}
	unlocking_technologies = {
		romanticism
	}
	cost = 125

	ai_weight = {
		value = 0
		if = {
			limit = {
				any_scope_building = {
					is_subsistence_building = yes
					occupancy < 0.75
				}
			}
			add = 10
		}
		if = {
			limit = {
				any_scope_building = {
					is_subsistence_building = yes
					occupancy < 0.5
				}
			}
			add = 15
		}
		if = {
			limit = {
				any_scope_building = {
					is_subsistence_building = yes
					occupancy < 0.25
				}
			}
			add = 25
		}
		if = {
			limit = {
				scope:country = {
					has_journal_entry = je_russification
				}
				culture_percent_state = {
					target = cu:russian
					value < je_russification_decree_russian_culture_target # set in event_values
				}
				OR = {
					is_in_geographic_region = geographic_region_finland_old
					is_in_geographic_region = geographic_region_baltic_states_old
					is_in_geographic_region = geographic_region_poland_old
				}
			}
			add = 250
		}
		if = {
			limit = {
				scope:country = {
					has_journal_entry = je_tp_tsar_heir_of_alexander
					NOT = { has_variable = je_tp_tsar_heir_of_alexander_button_1_complete }
				}
				state_region = {
					this = s:STATE_OUTER_MANCHURIA
				}
			}
			add = 500
		}
		if = {
			limit = {
				scope:country = {
					has_journal_entry = je_tp_pool_develop_frontier
				}
				OR = {
					is_in_geographic_region = geographic_region_siberia
					is_in_geographic_region = geographic_region_central_asia
					is_in_geographic_region = geographic_region_caucasus_old
				}
			}
			multiply = 1.5
		}
		if = {
			limit = {
				is_incorporated = no
				scope:country = { NOT = { has_law_or_variant = law_type:law_colonial_resettlement } } # Already gives a hefty migration bonus
			}
			multiply = 3.0
		}
		if = {
			limit = {
				state_unemployment_rate >= 0.1
			}
			multiply = 0
		}
	}
}
