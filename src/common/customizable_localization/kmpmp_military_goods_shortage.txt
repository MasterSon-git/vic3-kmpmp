# Read-only presentation: applied tiers belong to the JE, price tests to its country.
# No effects, variables or cached copies of gameplay state are used here.

kmpmp_shortage_applied_small_arms = {
    type = journal_entry
    random_valid = no
    text = {
        trigger = { has_modifier = kmpmp_shortage_of_small_arms_low }
        localization_key = kmpmp_shortage_row_small_arms_low
    }
    text = {
        trigger = { has_modifier = kmpmp_shortage_of_small_arms_medium }
        localization_key = kmpmp_shortage_row_small_arms_medium
    }
    text = {
        trigger = { has_modifier = kmpmp_shortage_of_small_arms_high }
        localization_key = kmpmp_shortage_row_small_arms_high
    }
    text = {
        localization_key = EMPTY_STRING
    }
}

kmpmp_shortage_price_small_arms = {
    type = country
    random_valid = no
    text = {
        trigger = {
            NOT = { kmpmp_shortage_price_available = { MARKET_GOOD = small_arms } }
        }
        localization_key = kmpmp_shortage_price_unavailable
    }
    text = {
        trigger = {
            kmpmp_has_goods_shortage_low = { MARKET_GOOD = small_arms }
        }
        localization_key = kmpmp_shortage_price_low
    }
    text = {
        trigger = {
            kmpmp_has_goods_shortage_medium = { MARKET_GOOD = small_arms }
        }
        localization_key = kmpmp_shortage_price_medium
    }
    text = {
        trigger = {
            kmpmp_has_goods_shortage_high = { MARKET_GOOD = small_arms }
        }
        localization_key = kmpmp_shortage_price_high
    }
    text = {
        localization_key = kmpmp_shortage_price_below_low
    }
}
