# root = country; scope:war is supplied by the engine. No randomness.
# Retain KMPMP's incorporated-state occupation tiers as signed support loss.
kmpmp_war_support_from_occupation = {
    value = kmpmp_enemy_occupation_incorporated_states
    save_temporary_value_as = kmpmp_occupied_fraction
    value = 0
    if = { limit = { scope:kmpmp_occupied_fraction <= 0 } value = 0 }
    else_if = { limit = { scope:kmpmp_occupied_fraction < 0.1 } value = -0.1 }
    else_if = { limit = { scope:kmpmp_occupied_fraction < 0.25 } value = -0.5 }
    else_if = { limit = { scope:kmpmp_occupied_fraction < 0.5 } value = -1.0 }
    else_if = { limit = { scope:kmpmp_occupied_fraction < 0.75 } value = -2.0 }
    else_if = { limit = { scope:kmpmp_occupied_fraction < 0.9 } value = -3.0 }
    else = { value = -10.0 }
}
